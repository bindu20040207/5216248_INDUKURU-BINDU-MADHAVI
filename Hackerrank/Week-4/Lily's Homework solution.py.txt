import os

def count_swaps(arr, sorted_arr):
    n = len(arr)
    index_map = {v: i for i, v in enumerate(arr)}
    visited = [False] * n
    swaps = 0

    for i in range(n):
        if visited[i] or arr[i] == sorted_arr[i]:
            continue
        cycle_size = 0
        j = i
        while not visited[j]:
            visited[j] = True
            j = index_map[sorted_arr[j]]
            cycle_size += 1
        if cycle_size > 0:
            swaps += cycle_size - 1
    return swaps
def lilysHomework(arr):
    asc_sorted = sorted(arr)
    desc_sorted = sorted(arr, reverse=True)
    return min(count_swaps(arr[:], asc_sorted), count_swaps(arr[:], desc_sorted))


if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    n = int(input().strip())

    arr = list(map(int, input().rstrip().split()))

    result = lilysHomework(arr)

    fptr.write(str(result) + '\n')

    fptr.close()
